หมวดที่ 2 การบริหารราชการเพื่อให้เกิดประโยชน์สุขของประชาชน
{
    มาตราที่ 7 {
        การบริหารราชการเพื่อประโยชน์สุขของประชาชน {
            คือ การปฏิบัติราชการที่มีเป้าหมายเพื่อให้เกิดความผาสุกและความเป็นอยู่ที่ดีของประชาชน 
            ความสงบและปลอดภัยของสังคมส่วนรวม 
            ตลอดจนประโยชน์สูงสุดของประเทศ

            
        }

    }

    มาตราที่ 8 {
        ในการบริหารราชการเพื่อประโยชน์สูงสดของประชาชน{
            ส่วนราชการจะต้องดำเนินการ โดย{
                ถือว่าประชาชนเป็นศูนย์กลาง ที่จะได้รับการบริการจากรัฐ

                    1. กำหนดภารกิจของรั๘และส่วนราชการต้องเป็นไปตามมาตราที่ 7
                    และสอดคล้องกับแนวนโยบายของรัฐและนโยบายของคณะรัฐมนตรีที่แถลงต่อสภา
                    
                    2. การปฏิบัติภารกิจของส่วนราชการต้องเป็นไปโดยซื่อสัตย์ สุจริต ตรวจสอบได้ และมุ่งให้เกิดประโยชน์แก่ประชาชนในทุกระดับ

                    3. ก่อนเริ่มดำเนินการ ส่วนราชการต้องจัดให้มีการศึกษาวิเคราะห์ผลดี และ ผลเสีย ให้ดีก่อน

                    4. ให้หน้าที่ของข้าราชการที่จะต้องคอยรับฟังความคิดเห็น และ ความพึงพอใจของสังคมโดยรวมและประชาชนผู้รับบริการ

                    5. ในกรณีที่เกิดปัญหาและอุปสรรคจากการดำเนินการ ให้ส่วนราชการดำเนินการแก้ไขปัญหาโดยเร็วที่สุด
                }
                [กำหนดให้สอดคล้อง → สุจริต  → ก่อนเริ่มต้องคิดให้ดี → รับฟังความคิดเห็น → หากเกิดปัญหาให้รีบแก้]
            }

หมวดที่ 3 การบริหารราชการเพื่อให้เกิดผลสัมฤทธิ์ต่อภารกิจของรัฐ
{
    มาตราที่ 9 ให้ส่วนราชการปฏิบัติ ดังนี้ 
    {
        1. ก่อนจะทำภารกิจ ส่วนราชการต้องจัดทำแผนปฏิบัติราชการไว้ล่วงหน้า
        
        2. ในการกำหนดแผนปฏิบัติราชการต้อง มีรายละเอียดขั้นตอน ระยะเวลาและงบประมาณที่ต้องใช้
        เป้าหมายภารกิจ ผลสัมฤทธิ์ ตัวชี้วัดความสำเร็จ

        3. ส่วนราชการต้องติดตามและประเมินผลการปฏิบัติราชการตามหลักเกณฑ์และวิธีการที่ส่วนราชการกำหนดขึ้น

        4. ในกรณีที่การปฏิบัติราชการหรือปฏิบัติตามแผนเกิดผลกระทบต่อประชาชน ให้เป็นหน้าที่ของส่วนราชการ
        ที่ต้องดำเนินการแก้ไขหรือบรรเทาผลกระทบนั้น หรือ เปลี่ยนแผนปฏิบัติราชการให้เหมาะสม

    }
    [กำหนดแผนล่วงหน้า → กำหนดรายละเอียด → ติดตามประเมินผล → ถ้ามีปัญหาก็แก้ไข ไม่ก็เปลี่ยนแผน]

    มาตราที่ 16 {
        ส่วนราชการจัดทำแผน เป็น แผน 5 ปี และต้องสอดคล้องกับ ยุทธศาสตร์ชาติ แผนแม่บท แผนการปฏิรูปประเทศ แผนพัฒนาเศรษฐกิจและสังคมแห่งชาติ
        นโยบายของคณะรัฐมนตรีที่แถลงต่อรัฐสภา และ แผนอื่นๆที่เกี่ยวข้อง 
    }}

หมวดที่ 4 การบริหารราชการอย่างมีประสิทธิภาพ
{
    มาตราที่ 20 {
        เพื่อให้การปฏิบัติราชการในส่วนราชการเป็นไปอย่างมีประสิทธิภาพ ให้ส่วนราชการกำหนดเป้าหมาย
        แผนการทำงาน ระยะเวลา และ งบประมาณที่จะต้องใช้ในแต่ละงานหรือโครงการ และต้องเผยแพร่ให้ข้าราชการและประชาชนทราบด้วย (ป้ายเขียว)
    }

    มาตราที่ 21 {
        ให้ส่วนราชการจัดทำบัญชีต้นทุนในงานบริการสาธารณะ แต่ละประเภทตามหลักเกณฑ์และวิธีการที่กรมบัญชีกลางกำหนด
    }

    มาตราที่ 22 {
        ให้สำนักงานคณะกรรมการการพัฒนาเศรษฐกิจและสังคมแห่งชาติ และ สำนักงบประมาณ
        ร่วมกันให้มีการประเมินความคุ้มค่าในการปฏิบัติภารกิจของรัฐที่ส่วนราชการดำเนินการอยู่
        เพื่อรายงานคณะรัฐมนตรี เพื่อเป็นแนวทางในการจัดตั้งงบประมาณส่วนราชการในปีต่อไป

        สำนักงานสภาพัฒนาเศรษฐกิจและสังคมแห่งชาติ ร่วมกับ สำนักงบ เป็นผู้ร่วมกันประเมินความคุ้มค่าในการปฏิบัติภารกิจของรัฐ
    }

    มาตราที่ 23 {
        ในการจัดซื้อหรือจัดจ้าง ให้ส่วนราชการดำเนินการ โดยเปิดเผยและเที่ยงธรรม โดยพิจารณาถึงประโยชน์และผลเสียทางอ้อม ราคาและประโยชน์ระยะยาว
        
        แต่ในกรณีที่วัตถุประสงค์ในการใช้เป็นเหตุให้ต้องคิดถึงคุณภาพและการดูแลรักษา ให้สามารถซื้อได้โดยไม่ต้องถือราคาต่ำสุดในการเสนอซื้อหรือจัดจ้างเสมอไป
    }

    มาตราที่ 24 {
        ในการปฏิบัติภารกิจใด ถ้าส่วนราชการจำเป็นต้องได้รับอนุญาต อนุมัติ หรือ ความเห็นชอบจากส่วนราชการอื่นที่กฎหมาย กฎ ระเบียบ ข้อบังคับ ประกาศ หรือ มติคณะรัฐมนตรีกำหนด
        ให้ส่วนราชการที่มีอำนาจสูงสุด อนุญาต อนุมัติ หรือ ให้ความเห็นชอบ ดังกล่าว 
        แจ้งผลการพิจาณาให้ส่วนราชการที่ยื่นคำขอ ทราบภายใน {
            15 วัน นับตั้งแต่ที่ได้ยื่นคำขอ
        }
    }

    มาตราที่ 26 {
        การสั่งราชการ ปกติให้ทำเป็นลายลักษณ์อักษร แต่ถ้าผู้บังคับบัญชามีความจำเป็นที่ไม่สามารถสั่งเป็นลายลักษณ์อักษรได้ในตอนนั้น
        จะสั่งราชการด้วยวาจาก็ได้ แต่คนรับคำสั่งต้องบันทึกคำพูดเป็นลายลักษณ์อักษร และ ได้ปฏิบัติราชการตามคำสั่งแล้ว
    }}

หมวดที่ 5 การลดขั้นตอนการปฏิบัติงาน
{
    มาตราที่ 27 {
        ให้ส่วนราชการจัดมีการกระจายอำนาจการตัดสินใจเกี่ยวกับการสั่งการอนุญาต การอนุมัติ
        การปฏิบัติราชการ หรือ การดำเนินการอื่นๆ ของผู้ดำรงตำแหน่ง ให้ผู้ดำรงตำแหน่งคนอื่นมีหน้าที่รับผิดชอบในการดำเนินการในเรื่องนั้นได้โดยตรง
        เพื่อให้เกิดความรวดเร็ว และ ลดขั้นตอนการปฏิบัติราชการ}

    มาตราที่ 29 {
        ในการปฏิบัติงานเกี่ยวกับการบริการประชาชน หรือ การติดต่อประสานงานระหว่างราชการด้วยกัน
        ให้ส่วนราชการแต่ละที่ จัดทำแผนภูมิขั้นตอนและระยะเวลา รวมทั้งรายละเอียดอื่นๆ ที่เกี่ยวข้องในแต่ละขั้นตอนเปิดเผยไว้
        ที่ ที่ทำการของส่วนราชการและในระบบเครือข่ายสารสนเทศของส่วนราชการ 
        เพื่อให้ประชาชนและผู้ที่เกี่ยวข้องเข้าตรวจดูได้ } 
    วรรคที่ 2 ของมาตรา 29 {
        การบริการประชาชนและการติดต่อประสานงานระหว่างส่วนราชการด้วยกันต้องทำโดยใช้ 
        แพลตฟอร์มดิจิทัลกลางที่สำนักงานพัฒนรัฐบาลดิจิทัล (องค์การมหาชน) กำหนดด้วย
        }}

หมวดที่ 6 การปรับปรุงภารกิจของส่วนราชการ
{
    มาตราที่ 33 {
        ให้ส่วนราชการจัดให้มีการทบทวนภารกิจของตนว่า ภารกิจใดมีความจำเป็น หรือ สมควรยกเลิก ปรับปรุง หรือ เปลี่ยนแปลงการดำเนินการต่อไปหรือไม่
        โดยคำนึกถึงยุทธศาสตร์ชาติ แผนแม่บท แผนการปฏิรูปประเทศ แผยพัฒนาเศรษฐกิจและสังคมแห่งชาติ
        นโยบายของรัฐบาล แผนอื่นๆ ที่เกี่ยวข้อง รวมถึงเงิน ด้วย}
    
    มาตราที่ 34 {
        ในกรณีที่มีการยกเลิก โอน หรือ รวมส่วนราชการใดทั้งหมดหรือบางส่วน{
            ห้ามไม่ให้จัดตั้งส่วนราชการที่มีภารกิจหรืออำนาจหน้าที่ ที่มีลักษณะเดียวกันหรือคล้ายคลึงกันกับส่วนที่ยกเลิก โอน หรือ รวมไป ขึ้นมาอีก{
                เว้นแต่ มีเหตุผล และ ความจำเป็นเพื่อรักษาความมั่นคงของรัฐหรือเศรษฐกิจของประเทศ หรือ รักษาผลประโยชน์ส่วนรวมของประชาชน
                และต้องได้รับความเห็นชอบจาก ก.พ.ร.}
        }}}

หมวดที่ 7 การอำนวยความสะดวกและการตอบสนองความต้องการของประชาชน
{
    มาตราที่ 37 {
        ในการปฏิบัติราชการที่เกี่ยวข้องกับการบริการประชาชนหรือติดต่อประสานงานในระหว่างส่วนราชการด้วยกัน
        {
            ให้ส่วนราชการกำหนดระยะเวลาแล้วเสร็จของงานแต่ละงานและประกาศให้ประชาชนและข้าราชการทราบ}}
    
        หากประชาชนต้องการติดต่อหลายส่วนราชการ ให้จัดตั้ง {
        ศูนย์บริการร่วม (one stop service) เพื่อให้ประชาชนยื่นคำร้องหรือติดต่อที่เดียว
        }

    มาตราที่ 38 {
        เมื่อส่วนราชการใดได้รับการติดต่อสอบถามเป็นหนังสือจากประชาชนหรือจากส่วนราชการด้วยกัน
        เกี่ยวกับงานที่อยู้ในอำนาจเจ้าหน้าที่ของส่วนราชการนั้น ให้เป็นหน้าที่ขอวส่วนราชการนั้น ที่จะต้องตอบคำถามหรือแจ้งการดำเนินการให้ทราบ
        {
            ภายใน 15 วัน หรือ ภายในกำหนดเวลาที่กำหนดไว้ตามมาตราที่ 37}}}

ระเบียบนายกรัฐมนตรี{
    ว่าด้วยการสร้างระบบบริหารกิจการบ้านเมืองและสังคมที่ พ.ศ. 2542

    มีหลัก ธรรมาภิบาล / ธรรมรัฐ (good governance) มี 6 หลัก {
        1. หลักนิติธรรม {
            เป็นการปฏิบัติตามหรือระเบียบกฎหมาย}
        2. หลักคุณธรรม {
            เป้นการยึดถือความถูกต้อง ความซื่อสัตย์ สุจริต}
        3. หลักโปร่งใส {
            เป็นการตรวจสอบได้}
        4. หลักการมีส่วนร่วม {
            เป็นการเปิดโอกาสให้แสดงความคิดเห็น}
        5. หลักความรับผิดชอบ {
            เป็นการตระหนักในหน้าที่และพร้อมแก้ไขความผิดพลาด}
        6. หลักความคุ้มค่า {
            เป็นการใช้ทรัพยากรให้คุ้มค่าและเกิดประโยชน์สูงสุด}
    }

    ถ้า ธรรมาภิบาล ตาม ก.พ.ร. (ปี2546 - 2562) มี 10 หลัก

    กลยุทธ์ในการสร้างระบบบริหาร ต้องปฏิรูป 3 ส่วนไปพร้อมๆ กัน คือ {
        ภาครัฐ ภาคธุรกิจเอกชน และ ภาคประชาชน}}
